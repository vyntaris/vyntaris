cmake_minimum_required(VERSION 3.21)
project(vwa LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets DBus Network)
qt_add_executable(vyntaris-vwa main.cpp)
qt_add_resources(vyntaris-vwa scripts PREFIX / FILES accent.js)
target_link_libraries(vyntaris-vwa PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::DBus Qt6::Network)
install(TARGETS vyntaris-vwa RUNTIME DESTINATION bin)
